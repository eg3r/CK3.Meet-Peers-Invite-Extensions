# Meet Peers - Invite Extensions: decisions

# Ends the Meet Peers cooldown right away, so the next Meet Peers can be planned.
# The cooldown is var:mpie_meet_peers_cooldown (set in the playdate.txt override's on_start),
# not the native activity cooldown: the engine enforces that one by itself and no effect can clear it.
# Planning the next Meet Peers starts a new cooldown.
# Cooldowns recorded natively before this mod version can't be reset, so the decision is not shown for them.
# Off by default: needs the mpie_meet_peers_cooldown_reset game rule or the console global variable
# (see mpie_meet_peers_cooldown_reset_enabled_trigger).
mpie_reset_meet_peers_cooldown_decision = {
	picture = { reference = "gfx/interface/illustrations/decisions/bp2_playdate.dds" }

	ai_check_interval = 0

	is_shown = {
		mpie_meet_peers_cooldown_reset_enabled_trigger = yes
		is_ai = no
		is_adult = no
		has_variable = mpie_meet_peers_cooldown
	}

	effect = {
		remove_variable = mpie_meet_peers_cooldown
		custom_tooltip = mpie_reset_meet_peers_cooldown_decision_effect_tt
		mpie_debug_log_effect = { MSG = MPIE_cooldown_reset_decision_taken }
	}

	ai_will_do = {
		base = 0
	}
}
