# Meet Peers - Invite Extensions: extra guest invite rules for activity_playdate
#
# root = host of the activity
# Rules only build the candidate pool. Every candidate is still filtered by the
# base can_be_activity_guest rule and the playdate's own can_be_activity_guest
# (child aged 4-13, healthy, free, not a hostage, in diplomatic range).

# Neighboring children: neighboring rulers who are children themselves, and the children of neighboring rulers
# Neighboring child rulers are added for player hosts always, for AI hosts only with the
# mpie_ai_invites_players game rule on (with it "Off", AI hosts invite as before).
mpie_invite_rule_neighbors_children = {
	effect = {
		if = { # Probe: does the AI run this default rule, and how often do AI hosts build a guest list?
			limit = {
				debug_only = yes
				is_ai = yes
			}
			save_scope_value_as = {
				name = mpie_debug_rule_neighbor_children
				value = mpie_debug_rule_neighbor_children_value
			}
			mpie_debug_log_effect = { MSG = MPIE_rule_neighbors_ran }
		}
		# Rulers of bordering foreign realms (same iterator as vanilla activity_invite_neighbouring_rulers)
		top_liege ?= {
			every_neighboring_and_across_water_top_liege_realm_owner = {
				if = {
					limit = {
						is_adult = no
						OR = {
							root = { is_ai = no }
							mpie_ai_invites_players_enabled_trigger = yes
						}
						NOT = { is_in_list = characters }
					}
					add_to_list = characters
				}
				every_child = {
					limit = { NOT = { is_in_list = characters } }
					add_to_list = characters
				}
			}
		}
		# Same-rank rulers bordering the host, e.g. a count next door in another duchy
		every_neighboring_and_across_water_realm_same_rank_owner = {
			if = {
				limit = {
					is_adult = no
					OR = {
						root = { is_ai = no }
						mpie_ai_invites_players_enabled_trigger = yes
					}
					NOT = { is_in_list = characters }
				}
				add_to_list = characters
			}
			every_child = {
				limit = { NOT = { is_in_list = characters } }
				add_to_list = characters
			}
		}
	}
}

# Nearby player children (priority 2, after the family; see documentation/03_ai_meet_peers_invites_to_player.md):
# player children near the host, or of its faith within ~one France, depending on the
# mpie_ai_invites_players game rule (mpie_player_child_in_invite_range_trigger).
# For AI hosts this puts the player on the guest list; for a player host it adds the other
# nearby player children in multiplayer (nobody in single player).
# Cheap for every host: it only looks at player characters.
mpie_invite_rule_nearby_player_children = {
	effect = {
		if = {
			limit = { mpie_ai_invites_players_enabled_trigger = yes }
			every_player = {
				limit = {
					this != root
					mpie_player_child_eligible_trigger = yes
					NOT = { is_in_list = characters }
					mpie_player_child_in_invite_range_trigger = { HOST = root }
				}
				add_to_list = characters
				save_scope_as = mpie_debug_invited_player
				root = { mpie_debug_log_effect = { MSG = MPIE_rule_nearby_player_added } }
			}
		}
	}
}

# Children of the host's faith, at any court in diplomatic range
mpie_invite_rule_same_faith_children = {
	effect = {
		if = { # Probe: this rule is in the playdate's opt-in rules; does the AI run those too?
			limit = {
				debug_only = yes
				is_ai = yes
			}
			mpie_debug_log_effect = { MSG = MPIE_rule_optional_ran_for_ai }
		}
		mpie_add_children_in_range_effect = { FILTER = mpie_same_faith_as_host_trigger }
	}
}

# Children of the opposite sex to the host, at any court in diplomatic range
mpie_invite_rule_opposite_sex_children = {
	effect = {
		mpie_add_children_in_range_effect = { FILTER = mpie_opposite_sex_of_host_trigger }
	}
}

# Children of the court and vassals: when a ruler plans Meet Peers for a child at their court
# (mpie_court_triggers.txt), the children living at that court (their own, wards, relatives), so the
# young host can be picked from the guests (special guest mpie_host_child) and their siblings come
# along, and the children of the ruler's direct vassals (counts and up for AI rulers, like vanilla's
# vassal rules). Vanilla's default rules only invite powerful vassals' children; vassals' children is
# opt-in. Nothing for a child host.
mpie_invite_rule_court_children = {
	effect = {
		if = {
			limit = { is_adult = yes }
			if = { # Probe: how many children this rule offers an AI ruler (debug mode)
				limit = {
					debug_only = yes
					is_ai = yes
				}
				save_scope_value_as = {
					name = mpie_debug_rule_court_children
					value = mpie_debug_rule_court_children_value
				}
				save_scope_value_as = {
					name = mpie_debug_rule_vassal_children
					value = mpie_debug_rule_vassal_children_value
				}
				mpie_debug_log_effect = { MSG = MPIE_rule_court_children_ran }
			}
			every_courtier = {
				limit = {
					is_adult = no
					NOT = { is_in_list = characters }
				}
				add_to_list = characters
			}
			every_vassal = {
				limit = {
					trigger_if = { # Like vanilla: AI rulers skip barons, to save performance
						limit = {
							root = { is_ai = yes }
						}
						highest_held_title_tier >= tier_county
					}
				}
				every_child = {
					limit = {
						is_adult = no
						NOT = { is_in_list = characters }
					}
					add_to_list = characters
				}
			}
			# The friends and crushes of the children at court: the guest list is built for the ruler, so
			# vanilla's friends and crushes rules would only invite the ruler's own
			every_courtier = {
				limit = { is_adult = no }
				every_relation = {
					type = friend
					limit = {
						is_adult = no
						NOT = { is_in_list = characters }
					}
					add_to_list = characters
				}
				every_relation = {
					type = crush
					limit = {
						is_adult = no
						NOT = { is_in_list = characters }
					}
					add_to_list = characters
				}
			}
		}
	}
}

# Family first: vanilla's close and extended family rules (vanilla Meet Peers has neither), for player hosts
# always and for AI hosts only with the mpie_ai_invites_players rule on, so "Off" leaves AI-hosted Meet Peers
# as in the base game. Only children pass the guest filters: siblings, cousins, nephews and nieces. For a
# ruler planning for a child at court, the ruler's children are the young host's siblings and the ruler's
# nephews and nieces are the host's cousins.
mpie_invite_rule_close_family = {
	effect = {
		if = {
			limit = { mpie_family_first_enabled_trigger = yes }
			every_close_family_member = {
				limit = {
					bannable_serving_diarch_trigger = no # as vanilla
					NOT = { is_in_list = characters }
				}
				add_to_list = characters
			}
		}
	}
}

mpie_invite_rule_extended_family = {
	effect = {

			limit = { mpie_family_first_enabled_trigger = yes }
			every_extended_family_member = {
				limit = {
					bannable_serving_diarch_trigger = no # as vanilla
					NOT = { is_in_list = characters }
				}
				add_to_list = characters
			}
		}
	}
}
